require "rlsl"

shader = RLSL.to_wgsl(:pulse) do
  uniforms { float :time }
  fragment do |coord, resolution, u|
    uv = coord / resolution
    wave = sin(u.time + uv.x * 6.28)
    vec3(uv.x, uv.y, wave * 0.5 + 0.5)
  end
end
